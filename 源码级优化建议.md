## 一、安全性优化

### 1.1 SQL 注入防护（中等风险）
**当前状态**：大部分查询使用了参数化查询，但存在一些潜在风险。

**问题代码**：
```python

```

**优化建议**：
```python
# 使用更安全的方式构建IN查询
def build_in_query(field_name, values):
    """安全构建IN查询"""
    placeholders = ','.join(['%s' for _ in values])
    return f'{field_name} IN ({placeholders})', values

# 使用示例

cursor.execute(f'DELETE FROM clients WHERE {where_clause}', params)
```

```python
from collections import OrderedDict
import threading

class ProcessMonitor:
    def __init__(self, config=None):
        # 使用OrderedDict并设置最大大小
        self.high_usage_tracker = OrderedDict()
        self.max_tracker_size = 1000  # 限制追踪器大小
        self.tracker_lock = threading.Lock()
    
    def _cleanup_tracker(self):
        """清理过期的追踪记录"""
        with self.tracker_lock:
            if len(self.high_usage_tracker) > self.max_tracker_size:
                # 删除最旧的记录
                while len(self.high_usage_tracker) > self.max_tracker_size // 2:
                    self.high_usage_tracker.popitem(last=False)
```

```python
def with_db(func):
    """装饰器：自动管理数据库连接"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        conn = None
        try:
            conn = get_db()
            kwargs['conn'] = conn
            kwargs['cursor'] = conn.cursor()
            result = func(*args, **kwargs)
            conn.commit()
            return result
        except Exception as e:
            if conn:
                conn.rollback()
            raise e
        finally:
            if conn:
                conn.close()
    return wrapper

# 使用示例
```

```python
from dataclasses import dataclass
from typing import Optional, Dict, Any

@dataclass
class ErrorResponse:
    """统一错误响应"""
    error: str
    details: Optional[Dict[str, Any]] = None
    code: str = "UNKNOWN_ERROR"
    
    def to_dict(self):
        result = {'error': self.error, 'code': self.code}
        if self.details:
            result['details'] = self.details
        return result

def handle_error(e: Exception, context: str = "") -> tuple:
    """统一错误处理"""
    error_response = ErrorResponse(
        error=str(e),
        code=type(e).__name__,
        details={'context': context}
    )
    
    # 记录日志
    logger.error(f"{context}: {e}", exc_info=True)

    if isinstance(e, ValidationError):
        return jsonify(error_response.to_dict()), 400
    elif isinstance(e, NotFoundError):
        return jsonify(error_response.to_dict()), 404
    elif isinstance(e, PermissionError):
        return jsonify(error_response.to_dict()), 403
    else:
        return jsonify(error_response.to_dict()), 500

# 使用示例
```

### 5.2 重试机制优化（低风险）
**当前状态**：重试机制简单，可能不够健壮。

**优化建议**：
```python
import time
from functools import wraps

def retry(max_attempts=3, delay=1, backoff=2, exceptions=(Exception,)):
    """重试装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt < max_attempts - 1:
                        sleep_time = delay * (backoff ** attempt)
                        logger.warning(f"尝试 {attempt + 1}/{max_attempts} 失败，{sleep_time}秒后重试: {e}")
                        time.sleep(sleep_time)
            raise last_exception
        return wrapper
    return decorator

# 使用示例
@retry(max_attempts=3, delay=1, backoff=2, exceptions=(pymysql.err.OperationalError,))
def execute_query(cursor, query, params):
    """带重试的查询执行"""
    cursor.execute(query, params)
    return cursor.fetchall()
```

```python
import logging
import json
from datetime import datetime

class StructuredFormatter(logging.Formatter):
    """结构化日志格式化器"""
    def format(self, record):
        log_entry = {
            'timestamp': datetime.utcnow().isoformat(),
            'level': record.levelname,
            'logger': record.name,
            'message': record.getMessage(),
            'module': record.module,
            'function': record.funcName,
            'line': record.lineno
        }
        
        if hasattr(record, 'extra_data'):
            log_entry['extra'] = record.extra_data
        
        if record.exc_info:
            log_entry['exception'] = self.formatException(record.exc_info)
        
        return json.dumps(log_entry, ensure_ascii=False)

# 配置日志
def setup_logging():
    """配置结构化日志"""
    logger = logging.getLogger('hwmon')
    logger.setLevel(logging.INFO)
    
    handler = logging.StreamHandler()
    handler.setFormatter(StructuredFormatter())
    logger.addHandler(handler)
```
